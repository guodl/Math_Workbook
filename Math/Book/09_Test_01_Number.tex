    \item Solve $12x - 2 = x+ 7$.
	    \begin{exercise} 
            \begin{envSolution}
		        Solution:\\[0.5cm]
                    $\begin{mArrow}
                                  &  12x - 2 = x+ 7     \\
							      &  12x - 2 -x = x+ 7 -x        \\
     						      &  11x - 2  =  7                     \\ 
             					  &  11x - 2 +2   =  7 +2         \\
								  &  11x    =  7 +2          \\
								  & x=\frac{9}{11}
					 \end{mArrow}$			   
                     \begin{shortsolution}  
				           $  \frac{9}{11} $
			         \end{shortsolution}				 
            \end{envSolution}
        \end{exercise}
