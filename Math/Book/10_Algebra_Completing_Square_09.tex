  \begin{stask}[resume]
    \task Express $x^2 + 6x +1$  in the form $(x+a)^2 + b$, where $a$ and $b$ are constants.
              \begin{envOut}[equal height group=hRevCmQ1]{bisAnswer}
			       $\begin{sArrow}
			              & x^2 + 6x + 1                                                                                  \\[10pt]
					     =& \uwave{x^2+2\times 3x +3^2} - 3^2 +1     \Arrow{\boxNote[bisAnswer]{ $a^2+2ab+b^2=(a+b)^2 $}} \\[10pt]
                         =& (x+3)^2 - 3^2    + 1                                                                          \\[10pt]
                         =& (x+3)^2 - 8
	                \end{sArrow}$
              \end{envOut}%

    \task Express $x^2 - 6x +1$  in the form $(x+a)^2 +b$, where $a$ and $b$ are constants.
              \begin{envOut}[equal height group=hRevCmQ1]{bisAnswer}
			       $\begin{sArrow}
			              & x^2 - 6x + 1                                                                                    \\[10pt]
					     =& \uwave{x^2 - 2\times 3x +3^2} - 3^2 +1     \Arrow{\boxNote[bisAnswer]{ $a^2-2ab+b^2=(a-b)^2 $}} \\[10pt]
                         =& (x - 3)^2 - 3^2    + 1                                                                          \\[10pt]
                         =& (x - 3)^2 - 8
	                \end{sArrow}$
              \end{envOut}%

  \end{stask}	

\end{boxA}

  \begin{stask}[resume]
    \task Express $4x^2 + 24x +1$  in the form $a(x+b)^2 +c$, where a, b and c are constants.
              \begin{envOut}[equal height group=hRevCmQ1]{bisAnswer}
			       $\begin{sArrow}
			              & 4x^2 + 24x +1                                                                                  \\[10pt]
                         =& 4(x^2+6x) +1                                                                                   \\[10pt]
					     =& 4(\uwave{x^2+2\times 3x +3^2} - 3^2)  -1   \Arrow{\boxNote[bisAnswer]{ $a^2+2ab+b^2=(a+b)^2 $}} \\[10pt]
                         =& 4(x+3)^2 - 4 \times 3^2  -1                                                                    \\[10pt]
 						 =& 4(x+3)^2 - 37
	                \end{sArrow}$
              \end{envOut}%
    \task Express $4x^2 + 16x +1$  in the form $(ax+b)^2 +c$, where a , b and c are constants.
              \begin{envOut}[equal height group=hRevCmQ1]{bAnswer}
			       $\begin{sArrow}
			              & 4x^2 + 24x +1                                 \Arrow{$4x^2= (2x)^2$}                              \\[10pt]
			             =& (\uwave{2x})^2 + 24x +1                                                                           \\[10pt]
                         =& \underbrace{(2x)^2 + 2 \times 2x \times 6 + 6^2}_{a^2 + 2 \times a \times b + b^2} - 6^2 +1 
                               						 \Arrow{\boxNote[bisAnswer]{$a^2+2ab+b^2=(a+b)^2 $}} \\[10pt]
					     =& (2x+6)^2 - 6^2  -1                                                                                \\[10pt]
 						 =& (2x+6)^2 - 37
	                \end{sArrow}$
              \end{envOut}%
  \end{stask}	

\end{boxA}

\begin{boxA}[styleTitleLeft]{Revision \thenskill - Completing the Square}
   \pdfbookmark[subsubsection]{Revision \thenskill - Completing the Square}{bkASQ01}    
    Use completing the square to find the exact solutions of the equations.

  \begin{stask}[resume]
    \task $9x^2-6x-11=0$
              \begin{envOut}[equal height group=hRevCmQ1]{bisAnswer}
			       $\begin{sArrow}
			             \uwave{9x^2-6x}-11        &= 0                    \Arrow{\boxNote[bisAnswer]{ $a^2-2ab+b^2=(a-b)^2 $}}  \\[10pt]
                         \uwave{9x^2-6x+1 - 1} -11 &= 0                                                                         \\[10pt]
					     \uwave{(3x-1)^2 -1} -11   &= 0                                                                         \\[10pt]
                         (3x-1)^2   -12            &= 0                                                                         \\[10pt]
 						 (3x-1)^2                  &= 12                                                                        \\[10pt]
 						  3x-1                     &= \pm \sqrt{12}        \Arrow{$\textnormal{Simplify} \quad\sqrt{12}$}                    \\[10pt]
						  3x-1                     &= \pm 2\sqrt{3}                                                             \\[10pt]
						  x                        &= \frac{1}{3} \pm \frac{2\sqrt{3}}{3} 
	                \end{sArrow}$
              \end{envOut}%
    \task $x^2-8x-34=0$
              \begin{envOut}[equal height group=hRevCmQ1]{bAnswer}
			       $\begin{sArrow} 
						x &= 4 \pm 5 \sqrt{2}                                 \\[10pt]
						x &= 5
	                \end{sArrow}$
              \end{envOut}%
  \end{stask}	

\end{boxA}
