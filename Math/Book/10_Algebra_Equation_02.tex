   \task \begin{exercise} John travels \qty{60}{\kilo\metre} at an average speed of $x$ km/h.\\
                          He then travels a further \qty{150}{\kilo\metre} at an average speed of $(x+20)$ km/h. \\
						  The time taken for the whole journey is 4 hours.\\
						  Find the speed John travels for the first \qty{60}{\kilo\metre} of the journey.\\
						  You must show all your working.
              \begin{envSolution}
			      Solution:
			      \begin{align*}
			        A \quad & \frac{60}{x}+\frac{150}{x+20}=4  \\[10pt]
					      & 60(x+20)+150x=4x(x+20)           \\  
                          & 30(x+20)+75x=2x(x+20) 	  	    \\
					      & 30x+600+75x=2x^2+40x             \\
                     & 0=2x^2-65x-600                   \\
 					 & 2x^2-65x-600=0                   \\
					 & (2x+15)(x-40)=0                  \\
					 & x=-7.5 \quad or \quad x=40       \\
					 & x=-7.5 (ignore)                  \\
					 & x=40                             
				 \end{align*}
				 \-\hspace{1cm}John travels at an average speed of 40km/h for the first 60km.\\
                 \begin{shortsolution}  
				       % $40km/h$
                 \end{shortsolution}
             \end{envSolution}
        \end{exercise}
