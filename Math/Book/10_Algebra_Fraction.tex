\begin{bxExample}{}
        \textbf{Example}\\
	        \-\hspace{1cm} Simplify $ \frac{b-a}{a-b} $

    \tcbline 
    Solution:\\
		\begin{itemize}\item[]
            $\begin{mArrow}
                & \frac{b-a}{a-b}              \Arrow[tikz={-stealth}]{\text{Factorise -1 from the numerator:} $\uwave{-(a-b)}$} \\[10pt]
                = & \frac{-(a-b)}{a-b}                                                                            \\[10pt]
                = & -1
            \end{mArrow}$
        \end{itemize}
		
\end{bxExample}

	                       $ \frac{1}{x-1}-\frac{x}{x^2-1}    $
             \begin{envSolution}
			      Solution:\\[0.5cm]
                     \-\hspace{1.5cm} $\begin{sArrow}
                              & \frac{1}{x-1}-\frac{x}{x^2-1}                     \\
     						= & \frac{1}{x-1)}-\frac{x}{(x+1)(x-1)}                 \\
                            = & \frac{(x+1)}{(x+1)(x-1)}-\frac{x}{(x+1)(x-1)}  \\
                            = & \frac{1}{(x+1)(x-1)}                       
						  \end{sArrow}$			   
                 \begin{shortsolution}  
				       $ \frac{1}{(x+1)(x-1)}  $

               \end{envSolution}

                   & \sqrt{\dfrac{9x+18}{4}}   \\[15pt]

				 = & \frac{3}{2} \times \sqrt{x+2}                         \\[10pt]  
				 = & \frac{3 \sqrt{x+2}}{2}
              \end{mArrow}$
