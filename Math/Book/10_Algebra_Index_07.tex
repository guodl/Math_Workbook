\begin{mtask}
    \task Simplify $ \left(\frac{x^8}{16x^{16}}\right)^{-\frac{3}{4}} $
              \begin{envOut}[height=9cm]{bAnswer}
			       $\begin{sArrow}
			             & \left(\frac{x^8}{16x^{16}} \right)^{-\frac{3}{4}}    \Arrow{$ \left(\frac{a}{b} \right)^{-n} = \left(\frac{b}{a} \right)^{n} $} \\[10pt]
                       = & \left(\frac{16x^{16}}{x^8} \right)^{\frac{3}{4}}                                                                                \\
					   = & \left(\frac{2^4 x^{16}}{x^8} \right)^{\frac{3}{4}}                                                                              \\
					   = & \frac{2^{4\times \frac{3}{4} } x^{16 \times \frac{3}{4}} }{x^{8 \times \frac{3}{4}} }                                           \\
					   = & \frac{2^{3} x^{12}} {x^6 }                                                                                                      \\
					   = & 8 x^6
	                \end{sArrow}$
              \end{envOut}
			  \begin{exercise}\begin{shortsolution}  
				       $8 x^6$
			  \end{shortsolution}\end{exercise}	
\end{mtask}
