\begin{tasksample}(3)
    \task[]  Solve $\sin{x}=\frac{\sqrt{3}}{2}$ for $  0\si{\degree} \leqslant x \leqslant 360\si{\degree}$.
              \begin{envOut}[equal height group=hRevF2]{bisAnswer}
			       $\begin{sArrow}
			             & \sin{x}= \frac{\sqrt{3}}{2}                  \\        
                         & x=\sin^{-1}(\frac{\sqrt{3}}{2})    \Arrow{ } \\[1cm]
						 & x=60 \si{\degree}                        \\
                         & x=180\si{\degree} - 60 \si{\degree}  	\\[1cm] 
						 & x=60 \si{\degree}                        \\
						 & x=120 \si{\degree}
	                \end{sArrow}$
              \end{envOut}

                         & x=\cos^{-1}(\frac{\sqrt{3}}{2})    \Arrow{ } \\[1cm]

						 & x=30 \si{\degree}                        \\

\begin{enumerate}[resume]
  \pdfbookmark[section]{Probability}{bkProbability} 
  \item Two students are selected from a group of $10$ boys and $8$ girls. Find the probability that\\
		a) they are both boys.\\
		b) one is a boy and one is a girl.\\
      \begin{envOut}[height=18cm]{bAnswer}
	       \-\hspace{0.3cm} (a)\\
	       \-\hspace{1cm} $\begin{sArrow}
	             \text{P(both boys)} &= \text{P(1st Student boy, 2nd Student boy)}                             \Arrow{$\ $}               \\[10pt]
                              &= \frac{10}{18} \times \frac{10-1}{18-1}                  \\
                              &= \frac{10}{18} \times \frac{9}{17}                                    \\
                              &= \frac{5}{17} 
	        \end{sArrow}$\\[1cm]
			\-\hspace{0.3cm} (b)\\
	        \-\hspace{1cm} $\begin{sArrow}
	             \text{P(one boy and one girl)} &= \text{P(1st Student boy, 2nd Student girl)}  \Arrow[jump=2]{$\ $}   \\
				                   &\quad\quad  + \text{P(1st Student girl, 2nd Student boy)}                  \\
                              &= \frac{10}{18} \times \frac{8}{18-1}  + \frac{8}{18} \times \frac{10}{18-1}               \\
                              &= \frac{80}{153} 
	        \end{sArrow}$
      \end{envOut}
	  \begin{shortsolution}  
	     $ \quad \quad \quad \quad$
 	  \end{shortsolution}
\end{enumerate}
