    \pdfbookmark[subsubsection]{Skill \thenskill \quad Manipulating Surds}{sub1234} 
    \textbf{Example}
	\begin{itemize}[leftmargin=0.5cm]
	     \item[] Simplify $\sqrt{80}$
	\end{itemize}
\tcbline 

\textbf{Solution:}

$\begin{mArrow}
       \qquad  & \sqrt{80}                                         \\[10pt]
	          =& \sqrt{16 \times 5}                                \\[10pt]
			  =& \sqrt{16} \times \sqrt{5}                         \\[10pt]
		      =& 4 \times \sqrt{5}                                 \\[10pt]
              =& 4 \sqrt{5}			  
\end{mArrow}$

\tcbline

\begin{boxA}[styleTitleLeft]{Skill    \thenskill \quad Rationalising the Denominator}
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Rationalising the Denominator}{sub1234}
    Radical expressions such as $3 + \sqrt{5}$ and $3 - \sqrt{5}$ which are identical expcept for opposing signs in the middle, are called radical conjugates.\\
	The radical conjugate of \uwave{$a+\sqrt{b}$} is \uwave{$a-\sqrt{b}$}.\\
    \textbf{Example}
	\begin{itemize}[leftmargin=0.5cm]
	     \item[] Simplify $\frac{8}{3-\sqrt{5}}$ 
	\end{itemize}
\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

\hspace{0.5cm}\quad \quad 
$\begin{mArrow}
       \qquad  & \frac{8}{3-\sqrt{5}}                                                                 \\[10pt]
	          =& \frac{\mask{3-\sqrt{5}}{8} \hfill \times (3+\sqrt{5})}{(3-\sqrt{5}) \hfill \times (3+\sqrt{5})}   \\[10pt]
			  =& \frac{8\times (3+\sqrt{5})}{ 3^2 - (\sqrt{5})^2}        \\[10pt]
			  =& \frac{8\times (3+\sqrt{5})}{ 4}                         \\[10pt]
			  =& 2 \times (3+\sqrt{5})                                   \\[10pt]
			  =& 6 + 2\sqrt{5}                                 
\end{mArrow}$

\tcbline

\end{boxA}
