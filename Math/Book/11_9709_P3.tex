\begin{boxSkill}{Skill \thenskill \quad Remainder Theorem}
    \pdfbookmark[subsubsection]{Remainder Theorem}{subP3Rem1}
    \noindent%
    When a polynomial $f(x)$ is divided by $(x-a)$, the remainder is $f(a)$.\\[1cm]
	\textbf{Example:}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Find the remainder when the polynomial $f(x)=x^4 - 7x -8$ is divided by $(x+1)$.	 
	\end{itemize}

	\textbf{Solution:}\\[2mm]
    \begin{itemize}[leftmargin=0.8cm]
	   \item[]  $\begin{mArrow}
                   f(-1) & =  (-1)^4 -7 \times (-1) -8           \\
                      & =  0                                     
	            \end{mArrow}$
	   \item[] Remainder is 0.
	\end{itemize}   
 	 
	
	\tcbline
	
    \textbf{Common Mistake:}\\[0.6cm]
	\begin{warningEnv}
	    $\begin{sArrow}
	        \qquad   & \qquad\qquad\qquad \\[10pt]
	  	             & \qquad\qquad\qquad                                                                        \\[10pt] 
                     & \qquad\qquad\qquad 				   
		 \end{sArrow}$		   
    \end{warningEnv}
\end{boxSkill}

\begin{mtask}

   \task  Solve the inequality $|3x +1| \leqslant |x -5| $.
              \begin{envSolution}[height=10cm]
			       $\begin{mArrow}
			             & |3x +1|   \leqslant |x -5|        \\
			             & |3x +1|^2 \leqslant |x -5|^2  \\
			             & (3x+1)^2  \leqslant (x -5)^2  \\
			             & (3x+1)^2 -  (x -5)^2 \leqslant 0 \\
			             & (3x+1 + x-5)(3x+1 - (x-5)  \leqslant 0 \\  
			             & (4x-4) (2x +6)  \leqslant 0   \\
			             &  -3 \leqslant x \leqslant 1  
	                \end{mArrow}$
              \end{envSolution}			  
			  \begin{exercise}\begin{shortsolution}  
				       $  -3 \leqslant x \leqslant 1  $
			  \end{shortsolution}\end{exercise}

   \task  Solve the inequality $|7x - a| > 2|x + 3a| $, where $a$ is a positive constant.
              \begin{envSolution}[height=10cm]
			       $\begin{mArrow}
			             & |7x - a| > 2|x + 3a|        \\
			             & |7x - a|^2 > (2|x + 3a|)^2  \\
			             & (7x - a)^2 > (2(x + 3a))^2  \\
			             & (7x - a)^2 > (2x + 6a)^2   \\
			             & (7x - a)^2 - (2x + 6a)^2 >0 \\  
			             & (7x - a + 2x + 6a)(7x -a -2x - 6a) >0  \\
			             &  (9x+5a)(5x-7a) >0 \\[10pt]
			             &  x< -\frac{5a}{9} \quad \text{or}\quad  x>\frac{7a}{5}  
	                \end{mArrow}$
              \end{envSolution}			  
			  \begin{exercise}\begin{shortsolution}  
				       $ x< -\frac{5a}{9} \quad  \text{or} \quad x>\frac{7a}{5}  $
			  \end{shortsolution}\end{exercise}
\end{mtask}
