    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Chain Rule}{skilla1}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=\sin 5x$.
	\end{itemize}
\tcbline 
    \textbf{Solution:}\\ \vspace{0.5cm} 
    \hspace{1cm}Let $u=5x, \quad y=\sin u$ \\

    $\begin{sArrow}
	       \qquad \frac{dy}{dx}  &=  \frac{dy}{du} \times \frac{du}{dx} \Arrow{\circled{1} Chain Rule 
		                                                                       \boxNote[bisAnswer]{$\frac{dy}{dx}=\frac{dy}{du} \times \frac{du}{dx}$} }\\[15pt]
		                         &=  \cos u \times 5                                                  \\[10pt]
					   		     &=  5 \cos u                                                         \\[10pt] 
   					             &=  5 \cos 5x                                          
    \end{sArrow}$
	
	\tcbline
	
    \textbf{Note:}\\[0.6cm]
	\begin{warningEnv}
         $\begin{sArrow}
	        \qquad y   &=   \sin 5x    \Arrow{Common Mistake} \\[10pt]
	  	           y'  &=   \cos 5x 
          \end{sArrow}$
    \end{warningEnv}
\end{boxSkill}

\begin{boxSkill}{Differentiation - Chain Rule}    
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Chain Rule}{skilla2}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=\sin (x^2-3x-4)$.
	\end{itemize}
\tcbline 

\textbf{Analysis:}\\[0.5cm] 

\hspace{1cm}Let $u=x^2-3x-4, \quad y=\sin u $ \\[10pt]

$\begin{sArrow}
       \qquad         \frac{dy}{dx}    &=  \frac{dy}{du} \times \frac{du}{dx} \Arrow{\circled{1} Chain Rule 
	                                                                                 \boxNote[bisAnswer]{$\frac{dy}{dx}=\frac{dy}{du} \times \frac{du}{dx}$} }\\[15pt]
	                                   &=  \cos u \times (2x-3)                                                 \\[10pt]
									   &=  (2x-3) \cos u                                                                 \\[10pt] 
							           &=  (2x-3) \cos (x^2-3x-4)                                          
\end{sArrow}$

\tcbline 

\textbf{Solution:}\\[0.6cm] 
$\begin{sArrow}
       \qquad         \frac{dy}{dx}    &= \cos (x^2-3x-4) \times (x^2-3x-4)'   \\[15pt]
	                                   &= \cos (x^2-3x-4) \times (2x-3)        \\[10pt]
							           &=  (2x-3) \cos (x^2-3x-4)                                          
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{sArrow}
        \qquad y   &=   \sin (x^2-3x-4) \Arrow{Common Mistake} \\[10pt]
		       y'  &=   \cos (x^2-3x-4) 
    \end{sArrow}$

	     \item[] Differentiate $y=e^{\frac{1}{x^2}}$.
	\end{itemize}
\tcbline 

\textbf{Solution:}\\ \vspace{0.5cm} 
\hspace{0.5cm}$y=e^{\frac{1}{x^2}} $ \\[10pt]
\hspace{0.5cm}$y=e^{x^{-2}}        $ \\[10pt]
\hspace{1cm}Let $u=x^{-2}, \quad y=e^u $ \\[10pt]

$\begin{sArrow}
       \qquad \frac{dy}{dx} &=  \frac{dy}{du} \times \frac{du}{dx} \Arrow{\circled{1} Chain Rule 
		                                                                  \boxNote{$\frac{dy}{dx}=\frac{dy}{du} \times \frac{du}{dx}$} }\\[15pt]
	                        &=  e^u \times (-2)x^{-3}                                                  \\[10pt]
							&=  (-2)x^{-3} \; e^u                                                      \\[10pt] 
							&=  -2x^{-3} \; e^{x^{-2}}                                               \\[10pt]  
                            &=  -2x^{-3} \; e^{\frac{1}{x^2}}							
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{sArrow}
        \qquad y   &=   e^{\frac{1}{x^2}}   \Arrow{Common Mistake}\\[10pt]
		       y'  &=   e^{\frac{1}{x^2}} 
    \end{sArrow}$
	
\end{warningEnv}

\end{boxSkill}
