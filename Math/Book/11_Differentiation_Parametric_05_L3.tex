\begin{boxSkill}{Skill    \thenskill \quad Differentiation - Chain Rule}    %Pend_Update
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Chain Rule}{sub1234} 
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=(2x+3)^{10}$.
	\end{itemize}
\tcbline 

\textbf{Analysis:}
\vspace{0.5cm} 

$\begin{sArrow}
             \-\hspace{1cm}    \text{Let}\ u = 2x + 3 &,                                                                                \\[15pt] 
			 \-\hspace{1cm}    y  &= (2x+3)^{10} \  = u^{10}                                                                 \\[15pt]
                    \frac{dy}{dx} &=  \frac{dy}{du}  \times \frac{du}{dx} \Arrow{\boxNote[bisAnswer]{\circled{1} \text{Chain Rule} } }  \\[10pt]
	                              &=  10 u^9         \times 2             \Arrow{$u = 2x+3$}                                                  \\[10pt]
							      &=  10 (2x+3)^9    \times 2                                                                \\[10pt]
				   			      &=  20 (2x+3)^9                                       
\end{sArrow}$

\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

$\begin{sArrow}
			 \-\hspace{2cm}    y  &= (2x+3)^{10}          \Arrow{\boxNote[bisAnswer]{Chain Rule: $\quad \frac{dy}{dx}= \frac{dy}{du} \times \frac{du}{dx}$ } }  \\[10pt]                                                  \\[15pt]
                               y' &= 10 (2x+3)^9 \times (2x+3)'                                                                   \\[10pt]
	                                    &= 10 (2x+3)^9 \times 2                                                                         \\[10pt]
				   			            &= 20 (2x+3)^9                                       
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{sArrow}
        \qquad y   &=   (2x+3)^{10}  \Arrow{\text{Common Mistake}}\\[10pt]
		       y'  &=   10(2x+3)^9 
    \end{sArrow}$
	
\end{warningEnv}

\end{boxSkill}

\begin{mtask}[resume]
   \task \begin{exercise} A curve has parametric equations\\
                                  $x=t^2 + \sin{2t} $ and $y=e^{\sin t} + (t^2+1)^3$.\\
                          Write expressions for $\frac{dy}{dx}$ and $\frac{d^2y}{dx^2}. $        %Pending1234
              \begin{envSolution}[height=12cm]
                  \textbf{Solution:}%\\[10pt]    %Pending1234
				  
                  Let $u=3x-5, y=u^{4}$%\\[10pt]

                  $\begin{mArrow}
                       \frac{dy}{dx} &=  \frac{dy}{du} \times \frac{du}{dx} \\[15pt]
	                                &=  4u^3 \times 3                       \\[10pt]
		   				            &=  4(3x-5)^3 \times 3                  \\[10pt]
							        &=  12(3x-5)^3
                  \end{mArrow}$
                 \begin{shortsolution}  
				       =$12(3x-5)^3$
			     \end{shortsolution}				 
               \end{envSolution}
         \end{exercise}
\end{mtask}
