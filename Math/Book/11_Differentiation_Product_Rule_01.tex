\begin{boxSkill}{Skill    \thenskill \quad Differentiation - Product Rule}    
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Product Rule}{sub2x2x3p1}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=x^2(x^3+1)$.
	\end{itemize}
\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

$\begin{sArrow}
       \qquad  y' &=  (x^2)' (x^3+1) + x^2 (x^3+1)' \Arrow{\circled{1} Product Rule \boxNote[bisAnswer]{$(uv)'=u'v+uv'$} }\\[10pt]
	              &=  2x(x^3+1)  + x^2 \times 3x^2                                                                             \\[10pt]
				  &=  2x^4+2x  + 3x^4                                                                                          \\[10pt]
				  &=  5x^4+2x
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{mArrow}
        \qquad y  &=  x^2 (x^3+1)          \Arrow{Common Mistake}\\[10pt]
		       y' &=  (x^2)' (x^3+1)'    \\
	              &=  2x \times 3x^2       \\
				  &=  \cdots 
    \end{mArrow}$
	
\end{warningEnv}

\end{boxSkill}

\begin{boxSkill}{Skill    \thenskill \quad Differentiation - Product Rule}    
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Product Rule}{sub2x2sinx}
	The product rule allows us to differentiate the product of two functions.\\
	It states that if $y=uv$, where $u$ and $v$ are functions of $x$,then\\
	          \boxNote[bisAnswer]{ $(uv)'=u'v+uv'$}    
	\vspace{1cm}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=x^2 \sin x$.
	\end{itemize}
\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

$\begin{sArrow}
     \qquad y' &= (x^2)' \sin x + x^2 (\sin x)' \Arrow{\circled{1} Product Rule \boxNote[bisAnswer]{$(uv)'=u'v+uv'$} } \\[10pt]
	           &= 2x     \sin x + x^2 \cos x     
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{sArrow}
        \qquad y  &=  x^2 sinx          \Arrow{Common Mistake} \\[10pt]
		       y' &=  (x^2)' (sinx)'                           \\
	              &=  2x cosx
    \end{sArrow}$
	
\end{warningEnv}

\end{boxSkill}

\begin{enumerate}[resume] 
    \setcounter{enumi}{600}
    \item \begin{exercise} Differentiate $y=(x^2-1)\cos {x}$ \\
        \begin{envSolution}
		   \begin{align*} 
               y  & = (x^2-1)\cos {x}                   \\ 
               y' & = (x^2-1)'\cos {x}+(x^2-1)(\cos x)'	\\
			      & = 2x \cos {x} +(x^2-1)(-\sin x)     \\
				  & = 2x \cos {x} -(x^2-1)\sin x
           \end{align*}		
                \begin{shortsolution}
  	                $2x \cos {x} -(x^2-1)\sin x$
                \end{shortsolution}
        \end{envSolution}%
    \end{exercise}


\end{enumerate}


\begin{boxSkill}{Skill    \thenskill \quad Differentiation - Product Rule}    
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Product Rule}{sub2xdssdnx}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y=(x^2+1)e^x$.
	\end{itemize}
\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

$\begin{sArrow}
       \qquad y' &= (x^2+1)'e^x +(x^2+1)(e^x)' \Arrow{\circled{1} Product Rule \boxNote[bisAnswer]{$(uv)'=u'v+uv'$} }\\[10pt]
	             &= 2xe^x + (x^2+1)e^x         \Arrow{\circled{2} Factorising  } \\[10pt]
                 &= (x^2+2x+1)e^x                                                \\[10pt]
				 &= (x+1)^2e^x
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{mArrow}
        \qquad y  &=  (x^2+1)e^x       \Arrow{Common Mistake}\\[10pt]
		       y' &=  (x^2+1)' (e^x)'  \\
	              &=  2xe^x
    \end{mArrow}$
	
\end{warningEnv}

\end{boxSkill}

			      & = 1 \times \ln x +x \times \frac{1}{x} \\
				  & = \ln x +1
           \end{align*}

                \end{shortsolution}

\begin{boxSkill}{Skill    \thenskill \quad Differentiation - Product Rule}    
    \pdfbookmark[subsubsection]{Skill \thenskill \quad Differentiation - Product Rule}{sub2xtfg}
    \textbf{Example}
	\begin{itemize}[leftmargin=0.8cm]
	     \item[] Differentiate $y = 2 x e^x \ln x$.
	\end{itemize}
\tcbline 

\textbf{Solution:}
\vspace{0.5cm} 

$\begin{sArrow}
       \qquad y' &= (2x e^x \ln x)'  \\  
                 &=	2 (x e^x \ln x)'   \Arrow{\circled{1} Product Rule \boxNote[bisAnswer]{$(uvw)'=u'vw+uv'w+uvw'$} }\\[10pt]
	             &= 2\left[(x)'e^x \ln x + x(e^x)'\ln x +xe^x(\ln x)' \right]                                        \\[10pt]
                 &= 2\left[ 1\times e^x \ln x + x e^x\ln x + x e^x \frac{1}{x}  \right]                              \\[10pt]
                 &=	2\left[ e^x \ln x + x e^x\ln x +  e^x \right]    		       \Arrow{\circled{2} Factorising  } \\[10pt]
				 &= 2(\ln x + x \ln x +1)e^x                                                                         \\[10pt]
				 &= 2(1 + \ln x + x \ln x)e^x
\end{sArrow}$

\tcbline

\textbf{Note:}\\[0.6cm]
\begin{warningEnv}
    $\begin{mArrow}
        \qquad y  &=  2 x e^x \ln x                    \Arrow{Common Misconception}\\[10pt]
		       y' &=   2 \times e^x \times \frac{1}{x}                              \\
	              &=  \cdots 
    \end{mArrow}$
	
\end{warningEnv}

\end{boxSkill}
