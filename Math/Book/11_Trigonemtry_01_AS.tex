\begin{mtask}[resume]
   \task \begin{exercise} Prove the identity $\sin^4 x - \cos^4 x \equiv \sin^2 x - \cos^2 x$.
        \begin{envSolution}[height fill]
            $\begin{mArrow}
                \sin^4 x - \cos^4 x 
	            & \equiv (\sin^2 x - \cos^2 x)(\sin^2 x + \cos^2 x)  \Arrow{Difference of squares formula} \\
                & \equiv (\sin^2 x - \cos^2 x)(1)                    \Arrow{Use $\sin^2 x + \cos^2 x = 1$} \\
                & \equiv \sin^2 x - \cos^2 x.                                         
            \end{mArrow}$
            \begin{shortsolution}
                Use the difference of squares formula and simplify using fundamental identities.
            \end{shortsolution}
        \end{envSolution}
    \end{exercise}
\end{mtask}

\begin{mtask}[resume]    
    \task \begin{exercise} Prove the identity $\cos^4 x - \sin^4 x \equiv \cos^2 x - \sin^2 x$.
            \begin{envSolution}[height fill]
                $\begin{mArrow}
                    \cos^4 x - \sin^4 x 
                    & \equiv (\cos^2 x - \sin^2 x)(\cos^2 x + \sin^2 x)    \Arrow{Difference of squares formula} \\
                    & \equiv (\cos^2 x - \sin^2 x) \times 1                \Arrow{Use $\cos^2 x + \sin^2 x = 1$} \\
                    & \equiv \cos^2 x - \sin^2 x                                           
                \end{mArrow}$
                \begin{shortsolution}
                    Apply the difference of squares formula and use the double-angle identity.
                \end{shortsolution}
            \end{envSolution}
          \end{exercise}

\end{mtask}
